main :: IO ()
checkBrackets :: String -> Bool
checkBrackets input = checkBrackets' input []

-- Вспомогательная рекурсивная функция с использованием стека
checkBrackets' :: String -> [Char] -> Bool
checkBrackets' [] stack = null stack
checkBrackets' (x:xs) stack
  | isOpeningBracket x = checkBrackets' xs (x:stack)
  | isClosingBracket x = case stack of
                           []       -> False
                           (y:ys) -> if isMatchingPair y x
                                     then checkBrackets' xs ys
                                     else False
  | otherwise = checkBrackets' xs stack

-- Проверка, является ли символ открывающей скобкой
isOpeningBracket :: Char -> Bool
isOpeningBracket c = c `elem` "({["

-- Проверка, является ли символ закрывающей скобкой
isClosingBracket :: Char -> Bool
isClosingBracket c = c `elem` ")}]"

-- Проверка, соответствуют ли открывающая и закрывающая скобки друг другу
isMatchingPair :: Char -> Char -> Bool
isMatchingPair '(' ')' = True
isMatchingPair '{' '}' = True
isMatchingPair '[' ']' = True
isMatchingPair _ _     = False

-- Функция для подсчета уникальных слов и их количества в тексте
wordCount :: String -> [(String, Int)]
wordCount text = sortBy (flip compare `on` snd) $ map (\word -> (head word, length word)) (group (sort (words text))) where 
	on :: (b -> b -> c) -> (a -> b) -> a -> a -> c
	on f_binary f_unary x y = f_binary (f_unary x) (f_unary y)
main2 = do
	contents <- readFile "D:\\Haskell\\input.txt"
	writeFile "D:\\Haskell\\output.txt" (show $ wordCount contents)


-- Функция выводящая отсортированные уникальные слова
unicueWords :: String -> [(String, Int)]
unicueWords text = filter (\i -> snd i == 1) $ map (\word -> (head word, length word)) (group (sort (words text))) where 
	on :: (b -> b -> c) -> (a -> b) -> a -> a -> c
	on f_binary f_unary x y = f_binary (f_unary x) (f_unary y)

main3 = do
	contents <- readFile "D:\\Haskell\\input.txt"
	writeFile "D:\\Haskell\\output.txt" (show $ map (fst) (unicueWords contents))

--Вставляет заданный элемент между всеми элементами списка
intersperse :: a -> [a] -> [a]
intersperse _ [] = []
intersperse _ [x] = [x]					--Чтобы elem не вставлялся в конце списка
intersperse elem (x:xs) = x : elem : intersperse elem xs


--Разбивает строку(на самом деле список) на подсписки по заданному элементу(границы)
explodeMy :: Eq a => a -> [a] -> [[a]]
explodeMy _ [] = []					--Список пуст - значит, так надо(возвращаем пустой список дабы нормально сформировать подсписки)
explodeMy elem list = explodeHelper list [] where	--Изначально acc пустой, ибо подсписки изначально пустые
	explodeHelper [] acc = [acc]			--Если дошли до конца списка - возвращаем аккумулятор
	explodeHelper (x:xs) acc
		|x == elem = acc : explodeHelper xs []	--Дошли до разделителя - результируем acc и продолжаем дальше
		|otherwise = explodeHelper xs (acc ++ [x]) --Иначе, добавляем текущий элемент в аккамулятор(текущий подсписок)


--Разбивает список по заданному условию
explodeBy :: (a -> Bool) -> [a] -> [([a],[a])]
explodeBy _ [] = []
explodeBy f xs = (takeWhile (not.f) xs, takeWhile f (dropWhile (not.f) xs)) : explodeBy f (dropWhile f (dropWhile (not.f) xs))


-- Функция транспонирования матрицы
transpose :: [[a]] -> [[a]]
transpose ([]:_) = []
transpose xss = map head xss : transpose (map tail xss)


--Функция группировки подряд идущих элементов
groupMy :: Eq a => [a] -> [[a]]
groupMy [] = []
groupMy (x:xs) = (x : takeWhile (== x) xs) : groupMy (dropWhile (== x) xs)


-- Функция, которая ищет все префиксы ("abc" == ["","a","ab","abc"])
initsMy :: [a] -> [[a]]
initsMy xs = [take n xs | n <- [0..length xs]]


-- Функция для поиска всех суффиксов
tailsMy :: [a] -> [[a]]
tailsMy xs = [drop n xs | n <- [0..length xs]]


-- Функция для поиска всех инфиксов						--ВОЗМОЖНО ТАК НЕЛЬЗЯ!!!!!(МОЖНО!!!)
infixesMy :: [a] -> [[a]]
infixesMy xs = [] : [drop i (take j xs) | i <- [0..length xs], j <- [i  + 1..length xs]]



-- Проверка на префикс
isPrefixOfMy :: Eq a => [a] -> [a] -> Bool
isPrefixOfMy x y = if x `elem` (initsMy y) then True else False


--Проверка на суффкис
isSuffixOfMy :: Eq a => [a] -> [a] -> Bool
isSuffixOfMy x y = if x `elem` (tailsMy y) then True else False


--Проверка на инфикс								--ВОЗМОЖНО ТАК НЕЛЬЗЯ!!!!!(МОЖНО!!!)
isInfixOfMy :: Eq a => [a] -> [a] -> Bool
isInfixOfMy x y = if x `elem` (infixesMy y) then True else False


--Куммулятивные суммы постфиксов [1,2,2,4] == [9,8,6,4]				--Возможно первый элемент не берем
cumSumPostfixMy :: Num a => [a] -> [a]
cumSumPostfixMy xs = map sum (tailsMy xs)


--Куммулятивные суммы всех префиксов
cumSumPrefixMy :: Num a => [a] -> [a]
cumSumPrefixMy xs = map sum (initsMy xs)


--Список разниц между парами подряд идущих чисел
diffMy :: [Int] -> [Int]
diffMy xs = zipWith (-) (tail xs) (xs)

--Сколько чисел в списке СТРОГО меньше каждого числа из списка
lastFuncEver :: [Int] -> [Int]
lastFuncEver xs = map (\x -> length $ filter (<x) xs) xs
